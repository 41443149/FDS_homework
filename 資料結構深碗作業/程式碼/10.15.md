#include<iostream>
using namespace std;
class GameCharacter
{
public:
    string name;
    int hp;
    int level;

    GameCharacter() {
        name = "";
        hp = 100;
        level = 1;
    }

    GameCharacter(string n, int h, int l){
        name = n; // 將 n 存入 name
        hp = h;   // 將 h 存入 hp
        level = l;// 將 l 存入 level
    }
    void damage(int value){// 受到傷害
        this->hp -= value;  // 1. hp 扣掉 value
        if (hp < 0) hp = 0; // 2. 如果 hp 小於 0，將 hp 設為 0
    }
   
    void heal(int value){// 回復生命值
        this->hp += value;      // 1. hp 加上 value
        if (hp > 100) hp = 100;// 2. 如果 hp 大於 100，將 hp 設為 100
    }
    
    void levelUp(){// 升級
        level++;        // 1. level 加 1
        this->hp = 100;// 2. hp 回復成 100
    }
    
    void show(){// 顯示角色資料
        cout << "名字 : " << name << endl;// 顯示 name、hp、level
        cout << "血量 : " << hp << endl;
        cout << "等級 : " << level << endl;
    }
};
int main()
{
    int hp, level;
    string name;
    cout << "輸入角色數量 :";
    int character_num;
    cin >> character_num;
    GameCharacter* player = new GameCharacter[character_num];//
    if (character_num > 100) cout << "輸入角色太多(不能超過100個)";
    for (int i = 0; i < character_num; i++) {
        cout << "輸入角色資料" << endl;
        cout << "名字 : ";
        cin >> name;
        cout << "血量 : ";
        cin >> hp;
        cout << "等級 : ";
        cin >> level;
        player[i]=GameCharacter(name, hp, level);//
    }
    cout << "技能選單(輸入數字)" << endl;
    cout << "0. 結束" << endl;
    cout << "1. 扣血" << endl;
    cout << "2. 回血" << endl;
    cout << "3. 升級" << endl;
    cout << "4. Show角色資料" << endl;
    cout << "5. Show全部角色" << endl;

    int skill,who;
    while (true) {
        cout << "使用技能:";
        cin >> skill;
        cout << "哪個角色使用技能擊中:";
        cin >> who;
        switch (skill)
        {
        case 0:
            return 0;
        case 1:
            player[who-1].damage(50);
            player[who-1].show();
            break;
        case 2:
            player[who-1].heal(20);
            player[who-1].show();
            break;
        case 3:
            player[who-1].levelUp();
            player[who-1].show();
            break;
        case 4:
            player[who-1].show();
            break;
        case 5:
            player[who-1].show();
            break;
        }
    }  
    return 0;
}
